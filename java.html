import java.util.Scanner;

public class DeliveryPackageCounter {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int[] packages = new int[12];
        int total = 0;
        int top = 0;     
        int count = 0;   

        for (int i = 0; i < 12; i++) {
            do {
                System.out.print("Enter packages delivered by rider " + i + ": ");
                packages[i] = input.nextInt();
                if (packages[i] < 0) {
                    System.out.println("Invalid! Enter a nonnegative number.");
                }
            } while (packages[i] < 0);

            total += packages[i];

            if (packages[i] > packages[top]) {
                top = i;
            }
            if (packages[i] >= 20) {
                count++;
            }
        }

        double average = total / 12.0;

        System.out.println("Total deliveries: " + total);
        System.out.println("Average deliveries: " + average);
        System.out.println("Top rider: Rider " + top + " with " + packages[top] + " packages");
        System.out.println("Riders with at least 20 deliveries: " + count);
    }
}
